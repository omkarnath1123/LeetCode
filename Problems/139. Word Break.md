Given a string s and a dictionary of strings wordDict, return true if s can be segmented into a space-separated sequence of one or more dictionary words.

Note that the same word in the dictionary may be reused multiple times in the segmentation.

 

Example 1:

Input: s = "leetcode", wordDict = ["leet","code"]
Output: true
Explanation: Return true because "leetcode" can be segmented as "leet code".

Example 2:

Input: s = "applepenapple", wordDict = ["apple","pen"]
Output: true
Explanation: Return true because "applepenapple" can be segmented as "apple pen apple".
Note that you are allowed to reuse a dictionary word.

Example 3:

Input: s = "catsandog", wordDict = ["cats","dog","sand","and","cat"]
Output: false

 

Constraints:

	
- 1 <= s.length <= 300

	
- 1 <= wordDict.length <= 1000

	
- 1 <= wordDict[i].length <= 20

	
- s and wordDict[i] consist of only lowercase English letters.

	
- All the strings of wordDict are unique.

**Solution:**

- [x] [[Hash Table]]

class Solution {
    public boolean wordBreak(String s, List<String> wordDict) {
        
        Trie trie = new Trie();
        for (String word : wordDict) trie.insert(word);
        return trie.isPresent(s);
    }
    class Trie {
        TrieNode head = new TrieNode();
        public void insert(String s) {
            insertUtil(s, 0, head);
        }
        private void insertUtil(String s, int index, TrieNode node) {
            if (index == s.length()) {
                node.isWord = true;
            } else {

                int pos = s.charAt(index) - 'a';
                if (node.next[pos] == null)
                    node.next[pos] = new TrieNode();
                insertUtil(s, index + 1, node.next[pos]);
            }
        }
        public boolean isPresent(String s) {
            Boolean[] cache = new Boolean[s.length()];
            return isPresentUtil(s, 0, head, cache);
        }
        private boolean isPresentUtil(String s, int index, TrieNode node, Boolean[] cache) {
            if (index == s.length()) return node.isWord;

            if (node.isWord) {
                if (cache[index] == null) cache[index] = isPresentUtil(s, index, head, cache);
                if (cache[index]) return true;
            }
            int pos = s.charAt(index) - 'a';
            if (node.next[pos] == null) return false;
            return isPresentUtil(s, index + 1, node.next[pos], cache);
        }
    }
    class TrieNode {
        TrieNode[] next = new TrieNode[26];
        boolean isWord = false;
        public TrieNode() {}
    }
}
