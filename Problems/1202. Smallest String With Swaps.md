You are given a string s, and an array of pairs of indices in the string pairs where pairs[i] = [a, b] indicates 2 indices(0-indexed) of the string.

You can swap the characters at any pair of indices in the given pairs any number of times.

Return the lexicographically smallest string that s can be changed to after using the swaps.

 

Example 1:

Input: s = "dcab", pairs = [[0,3],[1,2]]
Output: "bacd"
Explaination: 
Swap s[0] and s[3], s = "bcad"
Swap s[1] and s[2], s = "bacd"

Example 2:

Input: s = "dcab", pairs = [[0,3],[1,2],[0,2]]
Output: "abcd"
Explaination: 
Swap s[0] and s[3], s = "bcad"
Swap s[0] and s[2], s = "acbd"
Swap s[1] and s[2], s = "abcd"

Example 3:

Input: s = "cba", pairs = [[0,1],[1,2]]
Output: "abc"
Explaination: 
Swap s[0] and s[1], s = "bca"
Swap s[1] and s[2], s = "bac"
Swap s[0] and s[1], s = "abc"

 

Constraints:

	
- 1 <= s.length <= 10^5

	
- 0 <= pairs.length <= 10^5

	
- 0 <= pairs[i][0], pairs[i][1] < s.length

	
- s only contains lower case English letters.

**Solution:**

- [x] [[Hash Table]]

class Solution {
    public String smallestStringWithSwaps(String s, List<List<Integer>> pairs) {
        
        DisjointSet set = new DisjointSet();
        int index = 0;
        for (char ch : s.toCharArray()) {
            set.makeSet(new Alphabet(ch, index));
            index++;
        }

        for (List<Integer> pair : pairs) {
            Alphabet one = new Alphabet(s.charAt(pair.get(0)), pair.get(0)),
                     two = new Alphabet(s.charAt(pair.get(1)), pair.get(1));
            set.union(one, two);
        }

        Map<Alphabet, PriorityQueue<Alphabet>> groups = set.getGroups();
        StringBuilder sb = new StringBuilder();
        index = 0;
        for (char ch : s.toCharArray()) {
            Alphabet head = set.find(new Alphabet(ch, index));
            sb.append(groups.get(head).remove().val);
            index++;
        }
        return sb.toString();
    }
    class DisjointSet {
        Map<Alphabet, Node> nodes = new HashMap<>();
        public void makeSet(Alphabet val) {
            nodes.putIfAbsent(val, new Node(val));
        }
        private Node findSet(Node node) {
            if (node.parent == node)
                return node;
            node.parent = findSet(node.parent);
            return node.parent;
        }
        public Alphabet find(Alphabet val) {
            makeSet(val);
            return findSet(nodes.get(val)).val;
        }
        public void union(Alphabet one, Alphabet two) {
            makeSet(one);
            makeSet(two);
            Node left = findSet(nodes.get(one)),
                 right = findSet(nodes.get(two));
            if (left == right) return;

            if (left.rank < right.rank)
                left.parent = right;
            else if (right.rank < left.rank)
                right.parent = left;
            else {
                left.parent = right;
                right.rank++;
            }
        }
        public Map<Alphabet, PriorityQueue<Alphabet>> getGroups() {

            Map<Alphabet, PriorityQueue<Alphabet>> groups = new HashMap<>();
            for (Alphabet node : nodes.keySet()) {
                Alphabet head = find(node);
                groups.putIfAbsent(head, new PriorityQueue<>());
                groups.get(head).add(node);
            }
            return groups;
        }
    }
    class Node {
        Alphabet val;
        int rank = 0;
        Node parent;
        public Node(Alphabet val) {
            this.val = val;
            this.parent = this;
        }
    }
    class Alphabet implements Comparable<Alphabet> {
        char val;
        int index;
        public Alphabet(char val, int index) {
            this.val = val;
            this.index = index;
        }
        @Override
        public int hashCode() {
            return Objects.hash(val, index);
        }
        @Override
        public boolean equals(Object obj) {
            Alphabet that = (Alphabet) obj;
            return this.val == that.val && this.index == that.index;
        }
        @Override
        public int compareTo(Alphabet that) {
            return Character.compare(this.val, that.val);
        }
    }
}
