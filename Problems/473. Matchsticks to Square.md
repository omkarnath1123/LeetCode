You are given an integer array matchsticks where matchsticks[i] is the length of the ith matchstick. You want to use all the matchsticks to make one square. You should not break any stick, but you can link them up, and each matchstick must be used exactly one time.

Return true if you can make this square and false otherwise.

 

Example 1:

Input: matchsticks = [1,1,2,2,2]
Output: true
Explanation: You can form a square with length 2, one side of the square came two sticks with length 1.

Example 2:

Input: matchsticks = [3,3,3,3,4]
Output: false
Explanation: You cannot find a way to form a square with all the matchsticks.

 

Constraints:

	
- 1 <= matchsticks.length <= 15

	
- 1 <= matchsticks[i] <= 108

**Solution:**

- [x] [[Dynamic Programming]]

class Solution {
    Map<String, Boolean> cache;
    public boolean makesquare(int[] matchsticks) {
        
        cache = new HashMap<>();
        int sum = 0,
            len = matchsticks.length;
        for (int sticks : matchsticks)
            sum = sum + sticks;
        if (sum % 4 != 0) return false;
        return canMake(sum / 4, sum / 4, 0, matchsticks, new boolean[len]);
    }
    private boolean canMake(int size, int remainSize, int side, int[] matchsticks, boolean[] picked) {
        if (side == 4) return true;

        String key = getKey(picked);
        if (cache.containsKey(key))
            return cache.get(key);
        
        int len = matchsticks.length;
        for (int i = 0; i < len; i++)
            if (!picked[i] && matchsticks[i] <= remainSize) {
                picked[i] = true;
                if (matchsticks[i] == remainSize) {
                    if (canMake(size, size, side + 1, matchsticks, picked)) {
                        cache.put(key, true);
                        picked[i] = false;
                        return true;
                    }
                } else {
                    if (canMake(size, remainSize - matchsticks[i], side, matchsticks, picked)) {
                        cache.put(key, true);
                        picked[i] = false;
                        return true;
                    }
                }
                picked[i] = false;
            }
        
        cache.put(key, false);
        return false;
    }
    private String getKey(boolean[] picked) {
        StringBuilder sb = new StringBuilder();
        for (boolean pick : picked)
            sb.append(pick).append(", ");
        return sb.toString();
    }
}
