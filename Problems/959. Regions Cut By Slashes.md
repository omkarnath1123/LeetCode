An n x n grid is composed of 1 x 1 squares where each 1 x 1 square consists of a '/', '\', or blank space ' '. These characters divide the square into contiguous regions.

Given the grid grid represented as a string array, return the number of regions.

Note that backslash characters are escaped, so a '\' is represented as '\\'.

 

Example 1:

Input: grid = [" /","/ "]
Output: 2

Example 2:

Input: grid = [" /","  "]
Output: 1

Example 3:

Input: grid = ["/\\","\\/"]
Output: 5
Explanation: Recall that because \ characters are escaped, "\\/" refers to \/, and "/\\" refers to /\.

 

Constraints:

	
- n == grid.length == grid[i].length

	
- 1 <= n <= 30

	
- grid[i][j] is either '/', '\', or ' '.

**Solution:**

- [x] [[Hash Table]]

class Solution {
    char FORWARD = '/', BACK = '\\';
    public int regionsBySlashes(String[] grid) {
        
        DisjointSet set = new DisjointSet();
        int len = grid.length;
        for (int i = 0; i < len; i++) {

            String row = grid[i];
            for (int j = 0; j < len; j++) {
                char ch = row.charAt(j);

                Triangle up = new Triangle(Pos.UP, i, j),
                         left = new Triangle(Pos.LEFT, i, j),
                         down = new Triangle(Pos.DOWN, i, j),
                         right = new Triangle(Pos.RIGHT, i, j);
                if (ch == FORWARD) {
                    set.union(up, left);
                    set.union(down, right);
                } else if (ch == BACK) {
                    set.union(left, down);
                    set.union(up, right);
                } else {
                    set.union(up, left);
                    set.union(left, down);
                    set.union(down, right);
                }
            }
        }
        for (int i = 0; i < len; i++)
            for (int j = 0; j < len; j++) {
                if (i > 0) {
                    Triangle prev = new Triangle(Pos.DOWN, i - 1, j),
                             curr = new Triangle(Pos.UP, i, j);
                    set.union(prev, curr);
                }
                if (j > 0) {
                    Triangle prev = new Triangle(Pos.RIGHT, i, j - 1),
                             curr = new Triangle(Pos.LEFT, i, j);
                    set.union(prev, curr);
                }
            }

        return set.getHeadCount();
    }
    class DisjointSet {
        Map<Triangle, Node> nodes = new HashMap<>();
        private void makeSet(Triangle val) {
            nodes.putIfAbsent(val, new Node(val));
        }
        private Node findSet(Node node) {
            if (node.parent == node)
                return node;
            node.parent = findSet(node.parent);
            return node.parent;
        }
        public Triangle find(Triangle val) {
            makeSet(val);
            return findSet(nodes.get(val)).val;
        }
        public void union(Triangle one, Triangle two) {
            makeSet(one);
            makeSet(two);
            Node left = findSet(nodes.get(one)),
                 right = findSet(nodes.get(two));
            if (left == right) return;

            if (left.rank < right.rank)
                left.parent = right;
            else if (right.rank < left.rank)
                right.parent = left;
            else {
                left.parent = right;
                right.rank++;
            }
        }
        public int getHeadCount() {

            Set<Triangle> heads = new HashSet<>();
            for (Triangle tri : nodes.keySet())
                heads.add(findSet(nodes.get(tri)).val);
            return heads.size();
        }
    }
    class Node {
        Triangle val;
        Node parent;
        int rank = 1;
        public Node (Triangle val) {
            this.val = val;
            this.parent = this;
        }
    }
    class Triangle {
        Pos pos;
        int x, y;
        public Triangle(Pos pos, int x, int y) {
            this.pos = pos;
            this.x = x;
            this.y = y;
        }
        @Override
        public boolean equals(Object obj) {
            Triangle that = (Triangle) obj;
            return this.pos == that.pos && this.x == that.x && this.y == that.y;
        }
        @Override
        public int hashCode() {
            return Objects.hash(pos, x, y);
        }
    }
    enum Pos {
        UP, DOWN, LEFT, RIGHT;
    }
}
