Given a binary tree, determine if it is height-balanced.

 

Example 1:

Input: root = [3,9,20,null,null,15,7]
Output: true

Example 2:

Input: root = [1,2,2,3,3,null,null,4,4]
Output: false

Example 3:

Input: root = []
Output: true

 

Constraints:

	
- The number of nodes in the tree is in the range [0, 5000].

	
- -104 <= Node.val <= 104

**Solution:**

- [x] [[Depth First Search]]

/**
 * Definition for a binary tree node.
 * public class TreeNode {
 *     int val;
 *     TreeNode left;
 *     TreeNode right;
 *     TreeNode() {}
 *     TreeNode(int val) { this.val = val; }
 *     TreeNode(int val, TreeNode left, TreeNode right) {
 *         this.val = val;
 *         this.left = left;
 *         this.right = right;
 *     }
 * }
 */
class Solution {
    public boolean isBalanced(TreeNode root) {

      Pair res = traverse(root);
      return res.balanced;
    }
    private Pair traverse(TreeNode node) {
      if (node == null) return new Pair(true, 0);

      Pair left = traverse(node.left),
           right = traverse(node.right);

      return new Pair(left.balanced && right.balanced && (Math.abs(left.height - right.height) <= 1), Math.max(left.height, right.height) + 1);
    }
    class Pair {
      boolean balanced;
      int height;
      public Pair(boolean balanced, int height) {
        this.balanced = balanced;
        this.height = height;
      }
    }
}
