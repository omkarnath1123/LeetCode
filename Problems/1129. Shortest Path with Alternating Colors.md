You are given an integer n, the number of nodes in a directed graph where the nodes are labeled from 0 to n - 1. Each edge is red or blue in this graph, and there could be self-edges and parallel edges.

You are given two arrays redEdges and blueEdges where:

	
- redEdges[i] = [ai, bi] indicates that there is a directed red edge from node ai to node bi in the graph, and

	
- blueEdges[j] = [uj, vj] indicates that there is a directed blue edge from node uj to node vj in the graph.

Return an array answer of length n, where each answer[x] is the length of the shortest path from node 0 to node x such that the edge colors alternate along the path, or -1 if such a path does not exist.

 

Example 1:

Input: n = 3, redEdges = [[0,1],[1,2]], blueEdges = []
Output: [0,1,-1]

Example 2:

Input: n = 3, redEdges = [[0,1]], blueEdges = [[2,1]]
Output: [0,1,-1]

 

Constraints:

	
- 1 <= n <= 100

	
- 0 <= redEdges.length, blueEdges.length <= 400

	
- redEdges[i].length == blueEdges[j].length == 2

	
- 0 <= ai, bi, uj, vj < n

**Solution:**

- [x] [[Breadth First Search]]

class Solution {
    Map<Integer, List<Integer>> red, blue;
    public int[] shortestAlternatingPaths(int n, int[][] redEdges, int[][] blueEdges) {
        
        red = new HashMap<>();
        blue = new HashMap<>();

        for (int[] redEdge : redEdges) {
            red.putIfAbsent(redEdge[0], new ArrayList<>());
            red.get(redEdge[0]).add(redEdge[1]);
        }
        for (int[] blueEdge : blueEdges) {
            blue.putIfAbsent(blueEdge[0], new ArrayList<>());
            blue.get(blueEdge[0]).add(blueEdge[1]);
        }

        int[] path = new int[n];
        for (int i = 0; i < n; i++)
            path[i] = Integer.MAX_VALUE;

        Set<Pair> visited = new HashSet<>();
        Deque<Pair> queue = new ArrayDeque<>();

        Pair startRed = new Pair(0, true),
             startBlue = new Pair(0, false);
        visited.add(startRed);
        visited.add(startBlue);
        queue.offerLast(startRed);
        queue.offerLast(startBlue);

        int size = queue.size(), depth = 0;
        while (!queue.isEmpty()) {
            size--;
            Pair poll = queue.pollFirst();
            path[poll.node] = Math.min(path[poll.node], depth);

            List<Integer> nexts = poll.pickRed ? 
                                    red.getOrDefault(poll.node, new ArrayList<>()) : 
                                    blue.getOrDefault(poll.node, new ArrayList<>());
            for (int next : nexts) {
                Pair nextPair = new Pair(next, !poll.pickRed);
                if (visited.contains(nextPair)) continue;
                visited.add(nextPair);
                queue.offerLast(nextPair);
            }

            if (size == 0) {
                size = queue.size();
                depth++;
            }
        }

        for (int i = 1; i < n; i++)
            if (path[i] == Integer.MAX_VALUE)
                path[i] = -1;

        return path;
    }
    class Pair {
        int node;
        boolean pickRed;
        public Pair(int node, boolean pickRed) {
            this.node = node;
            this.pickRed = pickRed;
        }
        @Override
        public boolean equals(Object obj) {
            Pair that = (Pair) obj;
            return this.node == that.node && this.pickRed == that.pickRed;
        }
        @Override
        public int hashCode() {
            return Objects.hash(this.node, this.pickRed);
        }
    }
}
