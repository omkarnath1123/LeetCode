Given an unsorted array of integers nums, return the length of the longest consecutive elements sequence.

You must write an algorithm that runs in O(n) time.

 

Example 1:

Input: nums = [100,4,200,1,3,2]
Output: 4
Explanation: The longest consecutive elements sequence is [1, 2, 3, 4]. Therefore its length is 4.

Example 2:

Input: nums = [0,3,7,2,5,8,4,6,0,1]
Output: 9

Example 3:

Input: nums = [1,0,1,2]
Output: 3

 

Constraints:

	
- 0 <= nums.length <= 105

	
- -109 <= nums[i] <= 109

class Solution {
    public int longestConsecutive(int[] nums) {
        
        Set<Integer> nos = new HashSet<>();
        for (int num : nums) nos.add(num);

        DisjointSet set = new DisjointSet();
        for (int num : nums) {
            set.union(num, num);
            if (nos.contains(num - 1)) set.union(num, num - 1);
            if (nos.contains(num + 1)) set.union(num, num + 1);
        }
        return set.getMaxSizeGroup();
    }
    class DisjointSet {
        Map<Integer, Node> nodes = new HashMap<>();
        private void makeSet(int val) {
            nodes.putIfAbsent(val, new Node(val));
        }
        private Node findSet(Node node) {
            if (node.parent == node)
                return node;
            node.parent = findSet(node.parent).parent;
            return node.parent;
        }
        public int find(int val) {
            makeSet(val);
            return findSet(nodes.get(val)).val;
        }
        public void union(int one, int two) {
            makeSet(one);
            makeSet(two);
            Node left = findSet(nodes.get(one)),
                 right = findSet(nodes.get(two));
            if (left == right) return;

            if (left.rank < right.rank)
                left.parent = right;
            else if (left.rank > right.rank)
                right.parent = left;
            else {
                left.parent = right;
                right.rank++;
            }
        }
        public int getMaxSizeGroup() {

            Map<Integer, Integer> size = new HashMap<>();
            for (int node : nodes.keySet())
                size.merge(find(node), 1, Integer::sum);

            int maxSize = 0;
            for (int groupSize : size.values())
                maxSize = Math.max(maxSize, groupSize);
            return maxSize;
        }
    }
    class Node {
        int val, rank = 1;
        Node parent;
        public Node(int val) {
            this.val = val;
            this.parent = this;
        }
    }
}
