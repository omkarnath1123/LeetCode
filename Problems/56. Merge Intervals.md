Given an array of intervals where intervals[i] = [starti, endi], merge all overlapping intervals, and return an array of the non-overlapping intervals that cover all the intervals in the input.

 

Example 1:

Input: intervals = [[1,3],[2,6],[8,10],[15,18]]
Output: [[1,6],[8,10],[15,18]]
Explanation: Since intervals [1,3] and [2,6] overlap, merge them into [1,6].

Example 2:

Input: intervals = [[1,4],[4,5]]
Output: [[1,5]]
Explanation: Intervals [1,4] and [4,5] are considered overlapping.

Example 3:

Input: intervals = [[4,7],[1,4]]
Output: [[1,7]]
Explanation: Intervals [1,4] and [4,7] are considered overlapping.

 

Constraints:

	
- 1 <= intervals.length <= 104

	
- intervals[i].length == 2

	
- 0 <= starti <= endi <= 104

**Solution:**

- [x] [[Array]]

from enum import Enum

class State(Enum):
    START = 0
    END = 1

class Interval:

    def __init__(self, index: int, state: State):
        self.index = index
        self.state = state

    def __lt__(self, other: 'Interval'):
        if self.index == other.index:
            return self.state.value < other.state.value
        else:
            return self.index < other.index

class Solution:
    def merge(self, intervals: List[List[int]]) -> List[List[int]]:

        min_heap = []
        for interval in intervals:
            heapq.heappush(min_heap, Interval(interval[0], State.START))
            heapq.heappush(min_heap, Interval(interval[1], State.END))

        merged = []
        open_count = 0
        first_open = None
        while min_heap:

            poll = heapq.heappop(min_heap)

            if poll.state == State.START: open_count += 1
            else: open_count -= 1

            if open_count == 1 and first_open is None: first_open = poll
            if open_count == 0:
                merged.append([first_open.index, poll.index])
                first_open = None

        return merged

class Solution {
    public int[][] merge(int[][] intervals) {
        
        PriorityQueue<Event> minHeap = new PriorityQueue<>();
        for (int[] interval : intervals) {
            minHeap.add(new Event(interval[0], Pos.START));
            minHeap.add(new Event(interval[1], Pos.END));
        }

        List<int[]> merged = new ArrayList<>();
        int open = 0;
        Event firstOpen = null;
        while (!minHeap.isEmpty()) {

            Event poll = minHeap.remove();
            if (poll.pos == Pos.START) open++;
            else open--;

            if (open == 1 && firstOpen == null) firstOpen = poll;
            if (open == 0) {
                merged.add(new int[]{firstOpen.index, poll.index});
                firstOpen = null;
            }
        }
        
        int[][] res = new int[merged.size()][2];
        for (int i = 0; i < merged.size(); i++)
            res[i] = merged.get(i);
        return res;
    }
    class Event implements Comparable<Event> {
        int index;
        Pos pos;
        public Event(int index, Pos pos) {
            this.index = index;
            this.pos = pos;
        }
        @Override
        public int compareTo(Event that) {
            if (this.index == that.index)
                return this.pos.compareTo(that.pos);
            else
                return Integer.compare(this.index, that.index);
        }
    }
    enum Pos {
        START, END
    }
}
