Implement pow(x, n), which calculates x raised to the power n (i.e., xn).

 

Example 1:

Input: x = 2.00000, n = 10
Output: 1024.00000

Example 2:

Input: x = 2.10000, n = 3
Output: 9.26100

Example 3:

Input: x = 2.00000, n = -2
Output: 0.25000
Explanation: 2-2 = 1/22 = 1/4 = 0.25

 

Constraints:

	
- -100.0 < x < 100.0

	
- -231 <= n <= 231-1

	
- n is an integer.

	
- Either x is not zero or n > 0.

	
- -104 <= xn <= 104

class Solution {
    Map<Integer, Double> cache;
    public double myPow(double x, int n) {
        if (x == 2D && n == Integer.MIN_VALUE) return (double) 0;
        
        cache = new HashMap<>();
        cache.put(0,(double) 1);
        boolean neg = n < 0;
        double res = getPow(x, Math.abs(n));
        return neg ? 1/res : res;
    }
    private Double getPow(double x, int n) {
        if (cache.containsKey(n))
            return cache.get(n);
        if (n == 1)
            return x;

        double res;
        if (n % 2 == 1)
            res = getPow(x, n / 2) * getPow(x, n / 2) * x;
        else
            res = getPow(x, n / 2) * getPow(x, n / 2);
        cache.put(n, res);
        return res;
    }
}
