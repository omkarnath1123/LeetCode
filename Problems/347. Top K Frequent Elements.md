Given an integer array nums and an integer k, return the k most frequent elements. You may return the answer in any order.

 

Example 1:

Input: nums = [1,1,1,2,2,3], k = 2

Output: [1,2]

Example 2:

Input: nums = [1], k = 1

Output: [1]

Example 3:

Input: nums = [1,2,1,2,1,2,3,1,3,2], k = 2

Output: [1,2]

 

Constraints:

	
- 1 <= nums.length <= 105

	
- -104 <= nums[i] <= 104

	
- k is in the range [1, the number of unique elements in the array].

	
- It is guaranteed that the answer is unique.

 

Follow up: Your algorithm's time complexity must be better than O(n log n), where n is the array's size.

```python3
class Element:
    def __init__(self, value, frequency):
        self.value = value
        self.frequency = frequency

    def __lt__(self, other: 'Element'):
        return self.frequency < other.frequency


class Solution:
    def topKFrequent(self, nums: List[int], k: int) -> List[int]:

        freq = Counter(nums)
        min_heap = []
        for key, value in freq.items():
            heapq.heappush(min_heap, Element(key, value))
            if len(min_heap) > k:
                heapq.heappop(min_heap)

        return [element.value for element in min_heap]
```

```java
class Solution {
    public int[] topKFrequent(int[] nums, int k) {
        
        Map<Integer, Integer> freq = new HashMap<>();
        for (int num : nums)
            freq.merge(num, 1, Integer::sum);

        Comparator<Integer> compare = new Comparator<Integer>(){
            @Override
            public int compare(Integer one, Integer two) {
                return Integer.compare(freq.get(one), freq.get(two));
            }
        };
        PriorityQueue<Integer> minHeap = new PriorityQueue<>(compare);

        for (int num : freq.keySet()) {
            minHeap.add(num);
            if (minHeap.size() > k) minHeap.remove();
        }

        int[] res = new int[k];
        int iter = 0;
        while(!minHeap.isEmpty()) {
            res[iter] = minHeap.remove();
            iter++;
        }
        return res;
    }
}
```
