You are given an array of strings equations that represent relationships between variables where each string equations[i] is of length 4 and takes one of two different forms: "xi==yi" or "xi!=yi".Here, xi and yi are lowercase letters (not necessarily different) that represent one-letter variable names.

Return true if it is possible to assign integers to variable names so as to satisfy all the given equations, or false otherwise.

 

Example 1:

Input: equations = ["a==b","b!=a"]
Output: false
Explanation: If we assign say, a = 1 and b = 1, then the first equation is satisfied, but not the second.
There is no way to assign the variables to satisfy both equations.

Example 2:

Input: equations = ["b==a","a==b"]
Output: true
Explanation: We could assign a = 1 and b = 1 to satisfy both equations.

 

Constraints:

	
- 1 <= equations.length <= 500

	
- equations[i].length == 4

	
- equations[i][0] is a lowercase letter.

	
- equations[i][1] is either '=' or '!'.

	
- equations[i][2] is '='.

	
- equations[i][3] is a lowercase letter.

class Solution {
    String equals = "==", notEquals = "!=";
    public boolean equationsPossible(String[] equations) {
        
        DisjointSet set = new DisjointSet();
        for (String equation : equations) {

            char left = equation.charAt(0),
                 right = equation.charAt(3);
            String sign = equation.substring(1, 3);
            char leftHead = set.find(left),
                 rightHead = set.find(right);
            if (sign.equals(equals))
                set.union(left, right);
        }

        for (String equation : equations) {

            char left = equation.charAt(0),
                 right = equation.charAt(3);
            String sign = equation.substring(1, 3);
            char leftHead = set.find(left),
                 rightHead = set.find(right);
            if (sign.equals(notEquals) && leftHead == rightHead)
                return false;
        }
        return true;
    }
    class DisjointSet {
        Map<Character, Node> nodes = new HashMap<>();
        private void makeSet(char val) {
            nodes.putIfAbsent(val, new Node(val));
        }
        private Node findSet(Node node) {
            if (node.parent == node)
                return node;
            node.parent = findSet(node.parent);
            return node.parent;
        }
        public char find(char ch) {
            makeSet(ch);
            return findSet(nodes.get(ch)).val;
        }
        public void union(char one, char two) {
            makeSet(one);
            makeSet(two);
            Node left = findSet(nodes.get(one)),
                 right = findSet(nodes.get(two));
            if (left == right) return;

            if (left.rank < right.rank)
                left.parent = right;
            else if (right.rank < left.rank)
                right.parent = left;
            else {
                left.parent = right;
                right.rank++;
            }
        }
    }
    class Node {
        char val;
        int rank = 0;
        Node parent;
        public Node(char val) {
            this.val = val;
            this.parent = this;
        }
    }
}
