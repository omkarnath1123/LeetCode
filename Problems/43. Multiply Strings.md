Given two non-negative integers num1 and num2 represented as strings, return the product of num1 and num2, also represented as a string.

Note: You must not use any built-in BigInteger library or convert the inputs to integer directly.

 

Example 1:

Input: num1 = "2", num2 = "3"
Output: "6"

Example 2:

Input: num1 = "123", num2 = "456"
Output: "56088"

 

Constraints:

	
- 1 <= num1.length, num2.length <= 200

	
- num1 and num2 consist of digits only.

	
- Both num1 and num2 do not contain any leading zero, except the number 0 itself.

class Solution {
public static String multiply(String one, String two) {
    if (one.equals("0") || two.equals("0")) return "0";
    int len = two.length();
    StringBuilder main = new StringBuilder();
    for (int i = 0; i < 400; i++)
      main.append("0");

    for (int i = len - 1; i >= 0; i--) {
      StringBuilder mul = multiplyOne(one, Integer.parseInt(two.charAt(i) + ""));
      sumTwo(main, mul, len - i - 1);
    }

    int index = 0;
    for (; index < main.length(); index++)
      if (main.charAt(index) != '0') break;
    return main.toString().substring(index);
  }

  private static StringBuilder multiplyOne(String s, int no) {

    StringBuilder sb = new StringBuilder();
    int len = s.length();
    int carray = 0;
    int num, sum;

    for (int i = len - 1; i >= 0; i--) {
      num = Integer.parseInt(s.charAt(i) + "");
      sum = num * no + carray;
      sb.insert(0, sum % 10);
      carray = sum / 10;
    }
    if (carray != 0) sb.insert(0, carray);
    return sb;
  }

  private static void sumTwo(StringBuilder total, StringBuilder sum, int shift) {

    boolean carray = false;
    int len = total.length(),
            lenSum = sum.length(),
            index = lenSum - 1,
            add, i;

    for (i = len - shift - 1; i >= 0 && index >= 0; i--, index--) {

      add = Integer.parseInt(total.charAt(i) + "") + Integer.parseInt(sum.charAt(index) + "") + (carray ? 1 : 0);
      total.setCharAt(i, Integer.toString(add % 10).charAt(0));
      carray = false;
      if (add / 10 == 1) carray = true;
    }
    if (carray) total.setCharAt(i, Integer.toString(1).charAt(0));
  }
}
